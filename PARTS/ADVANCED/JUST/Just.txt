    >> $dayOfMonth:"$dob" >> 1 to 31
    >> $dayOfWeek:"$dob" >> 1 to 7
    >> $dateTrunc:{
        start:"$dob",
        unit:"day" // day, month, year
    }
    >> $dateDiff:{
        startDate:"$dob",
        endDate:"$$NOW",
        unit:"day", // day, month, year
    }
    >> $dateToString:{
        format:"%y-%m",date:"$dob"
    }

        {
            $project: {
                created_at_plus_1_day: {

                    unit: "day",
                    amount: 1
                    }
                }
            }
        }
        ])
